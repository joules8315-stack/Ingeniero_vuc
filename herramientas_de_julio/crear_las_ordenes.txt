def enc(que, como, donde, porque, extra=""):
    t = ("QUE SE REPARA: %s\nCOMO SE REPARA: %s\nDONDE SE REPARA: %s\nPOR QUE SE REPARA: %s"
         % (que, como, donde, porque))

def tabla_md(donde_plan):
    L = ["| # | Fallo | Evidencia medida | Causa raiz | Reparacion raiz | Donde | En que plan estaba | Ordenes | Paso | Que destraba | Meta |",
         "|---|---|---|---|---|---|---|---|---|---|---|"]
    for n, h in enumerate(HALLAZGOS, 1):
        plan = donde_plan(h["ordenes"]) + (" | ley: " + h["ley"] if h.get("ley") else "")
        fila = [h["fallo"], h["evidencia"], h["causa"], h["reparacion"], h["donde"], plan, h["ordenes"], h["paso"], h["destraba"], h["meta"]]
        L.append("| %d | %s |" % (n, " | ".join(x.replace("|", "/") for x in fila)))
    return "\n".join(L)


def flujo_md():

    for e in FLUJO:
        L.append("| " + " | ".join(x.replace("|", "/") for x in e) + " |")
    return "\n".join(L)


def metricas_md():
    L = ["| Termino | Medida | Hoy | Meta | Quien la mueve |", "|---|---|---|---|---|"]
    for m in METRICAS:
        L.append("| " + " | ".join(x.replace("|", "/") for x in m) + " |")
    return "\n".join(L)


def sha(ruta):
    import hashlib
    return hashlib.sha256(open(ruta, "rb").read()).hexdigest()[:16]



def medir_en_copia():
    """Corre en una COPIA (nunca en la carpeta real) las vigias que solo se podian medir en el PC de Julio."""
    import tempfile
    vig = ["vigias/test_vigia_el_paquete_trae_el_archivo_nombrado.py", "vigias/test_vigia_el_trabajo_pagado_no_se_pisa.py"]
    vig = [v for v in vig if os.path.isfile(os.path.join(RAIZ, v))]
    if not vig:
        return
    tmp = tempfile.mkdtemp(prefix="vuc_copia_")
    copia_ = os.path.join(tmp, "copia")
    try:
        shutil.copytree(RAIZ, copia_, ignore=shutil.ignore_patterns(".git", "__pycache__", "trabajos_del_equipo", "trabajos_sin_revisar", "rescate", "EXPEDIENTES", "planes_anulados"))
        cfg = os.path.join(copia_, "proyectos.config")
        lineas = open(cfg, encoding="utf-8", errors="replace").read().splitlines()
        lineas = [("ingeniero      = %s | main | ingeniero.py" % copia_) if re.match(r"^ingeniero\s*=", l_) else l_ for l_ in lineas]
        open(cfg, "w", encoding="utf-8").write("\n".join(lineas))
        print("\nVIGIAS QUE SOLO SE MEDIAN EN EL PC (se corren en una copia, la carpeta real no se toca):")
        for v in vig:
            r = subprocess.run([sys.executable, "-m", "pytest", v, "-q", "--tb=line", "-p", "no:cacheprovider", "-x"], cwd=copia_,
                               capture_output=True, text=True, timeout=180)
            sal = [l_ for l_ in (r.stdout + r.stderr).splitlines() if l_.strip()]
            print("   %-60s %s" % (os.path.basename(v)[12:-3][:60], "VERDE" if r.returncode == 0 else "ROJA: " + (sal[-1][:90] if sal else "")))
    except Exception as e_:
        print("   no se pudo medir en copia:", str(e_)[:120])
    finally:
        shutil.rmtree(tmp, ignore_errors=True)
